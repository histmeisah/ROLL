```python
@torch.no_grad()
def generate_sequences_with_search(self, prompts: DataProto, **kwargs) -> DataProto:
    # 提取输入和元数据
    input_ids = prompts.batch["input_ids"]
    attention_mask = prompts.batch["attention_mask"]
    position_ids = prompts.batch["position_ids"]
    device = input_ids.device
    
    # 设置最大轮数
    max_turns = min(self.max_turns, 3) if is_validate else self.max_turns
    
    # 跟踪哪些样本需要继续处理
    active_mask = torch.ones(expected_final_batch_size, dtype=torch.bool, device=device)
    final_answers = [None] * expected_final_batch_size
    final_responses = ['' for _ in range(expected_final_batch_size)]
    
    # 主生成循环
    for turn in range(max_turns):
        if not active_mask.any():
            print(f"[DEBUG] All samples completed at turn {turn}")
            break
        
        # 1. 准备当前轮次的输入
        active_prompts = DataProto.from_dict({
            'input_ids': fixed_size_input_ids[active_mask],
            'attention_mask': attention_mask[active_mask],
            'position_ids': position_ids[active_mask],
        })
        
        # 2. 调用模型生成响应
        print(f"\033[31mStart policy generating response...\033[0m")
        gen_output = super().generate_sequences(active_prompts, **{**local_kwargs, **search_kwargs})
        responses = gen_output.batch['responses']
        print(f"\033[31mEnd policy generating response...\033[0m")
        
        # 3. 处理搜索查询并执行搜索
        need_continue, search_results, turn_answers = self._process_search_in_batch(responses)
        
        # 4. 整合搜索结果到下一轮对话
        if active_mask.any() and turn < max_turns - 1:
            next_inputs = self._create_next_turn_inputs(
                fixed_size_input_ids[active_mask],
                turn_responses[active_mask],
                active_search_results,
                active_evaluation_results,
                active_summary_results
            )
    
    return result
```

#### 2.3 搜索处理函数: `_process_search_in_batch`

**位置**: `research/Agentic_LLM-search/src/workers/rollout/vllm_rollout/vllm_search_rollout_spmd.py:886`

这个函数负责解析模型响应中的搜索请求并执行批量搜索：

```python
def _process_search_in_batch(self, responses):
    """处理一批响应，识别并执行搜索"""
    batch_size = responses.size(0)
    
    # 解码响应文本
    response_texts = self.tokenizer.batch_decode(responses, skip_special_tokens=True)
    
    # 处理每个响应，查找搜索查询或答案
    search_queries = []
    need_continue = [False] * batch_size
    search_results_mapping = {}
    final_answers = [None] * batch_size
    
    for i, text in enumerate(response_texts):
        # 查找搜索标签 <search>query</search>
        search_match = re.search(self.search_pattern, text, re.DOTALL)
        # 查找答案标签 <answer>answer</answer>
        answer_match = re.search(self.answer_pattern, text, re.DOTALL)
        
        if search_match:
            # 提取搜索查询
            query = search_match.group(1).strip()
            search_queries.append(query)
            search_results_mapping[i] = len(search_queries) - 1
            need_continue[i] = True
        elif answer_match:
            # 提取最终答案
            final_answers[i] = answer_match.group(1).strip()
            need_continue[i] = False
    
    # 执行批量搜索
    search_results = {}
    if search_queries:
        print(f" start searching...")
        # 调用搜索客户端
        results = self.search_client.batch_search(search_queries)
        # 格式化搜索结果
        formatted_results = self.search_client.format_search_results(results)
        
        # 将结果映射回原始批次索引
        for batch_idx, result_idx in search_results_mapping.items():
            if result_idx < len(formatted_results):
                search_results[int(batch_idx)] = formatted_results[result_idx]
    
    return need_continue, search_results, final_answers
```

```python
class HTTPSearchClient:
    """HTTP客户端，用于远程搜索API调用"""
    
    def __init__(self, search_url, topk=3):
        self.search_url = search_url  # "http://127.0.0.1:8100/retrieve"
        self.topk = topk
    
    def batch_search(self, queries: List[str]):
        """调用远程搜索API并返回搜索结果"""
        if not queries:
            return []
        
        payload = {
            "queries": queries,
            "topk": self.topk,
            "return_scores": True
        }
        
        # 发送POST请求到检索服务器
        response = requests.post(self.search_url, json=payload, timeout=10).json()
        return response["result"]
    
    def format_search_results(self, results):
        """格式化搜索结果为可读文本"""
        formatted_results = []
        for result_set in results:
            format_reference = ''
            for idx, doc_item in enumerate(result_set):
                content = doc_item['document']['contents']
                title = content.split("\n")[0]
                text = "\n".join(content.split("\n")[1:])
                format_reference += f"Doc {idx + 1}(Title: {title}) {text}\n"
            formatted_results.append(format_reference)
        return formatted_results
```

```python
class DenseRetriever(BaseRetriever):
    def __init__(self, config):
        # 加载FAISS索引
        self.index = faiss.read_index(self.index_path)
        if config.faiss_gpu:
            # 将索引转移到GPU
            self.index = faiss.index_cpu_to_all_gpus(self.index)
        
        # 加载语料库
        self.corpus = load_corpus(self.corpus_path)
        
        # 初始化编码器
        self.encoder = Encoder(
            model_name=self.retrieval_method,
            model_path=config.retrieval_model_path,
            pooling_method=config.retrieval_pooling_method,
            max_length=config.retrieval_query_max_length,
            use_fp16=config.retrieval_use_fp16,
        )
    
    def _search(self, query: str, num: int = None, return_score: bool = False):
        # 编码查询
        query_emb = self.encoder.encode(query)
        
        # 在索引中搜索
        scores, idxs = self.index.search(query_emb, k=num)
        
        # 加载相关文档
        results = load_docs(self.corpus, idxs[0])
        
        return results, scores[0].tolist() if return_score else results
```

##### 编码器实现
```python
class Encoder:
    def __init__(self, model_name, model_path, pooling_method, max_length, use_fp16):
        self.model, self.tokenizer = load_model(model_path=model_path, use_fp16=use_fp16)
    
    @torch.no_grad()
    def encode(self, query_list: List[str], is_query=True) -> np.ndarray:
        # 为不同模型添加特定前缀
        if "e5" in self.model_name.lower():
            if is_query:
                query_list = [f"query: {query}" for query in query_list]
            else:
                query_list = [f"passage: {query}" for query in query_list]
        
        # 编码查询文本
        inputs = self.tokenizer(query_list, max_length=self.max_length, 
                               padding=True, truncation=True, return_tensors="pt")
        inputs = {k: v.cuda() for k, v in inputs.items()}
        
        # 获取文本嵌入
        output = self.model(**inputs, return_dict=True)
        query_emb = pooling(output.pooler_output, output.last_hidden_state,
                           inputs['attention_mask'], self.pooling_method)
        
        return query_emb.detach().cpu().numpy()
```

### 2. 关键技术组件

#### 2.1 搜索标签系统
- `<search>query</search>`: 触发搜索请求
- `<answer>answer</answer>`: 标记最终答案
- `<information>content</information>`: 包装搜索结果

#### 2.2 多轮对话管理
- `active_mask`: 跟踪哪些样本需要继续处理
- `turn_count`: 记录每个样本的对话轮数
- `max_turns`: 限制最大对话轮数

#### 2.3 状态掩码机制
- `info_mask`: 在训练时掩码检索信息
- `evaluation_mask`: 掩码评估信息
- 确保模型只学习推理过程，不学习检索内容

### 3. 性能优化特性

#### 3.1 批量处理
- 支持批量搜索请求，减少网络开销
- 并行处理多个查询，提高效率

#### 3.2 GPU加速
- FAISS GPU版本加速索引搜索
- E5模型GPU推理加速编码

#### 3.3 内存管理
- 动态批次大小调整
- 及时释放GPU内存

```bash
--retriever_model: E5模型路径
--faiss_gpu: 启用GPU加速
```

这个检索搜索系统实现了一个完整的检索增强生成(RAG)框架，具有以下特点：

1. **完整的调用链路**: 从shell脚本到核心检索代码的完整实现

4. **灵活的配置**: 支持多种后端和配置选项
5. **状态管理**: 完善的对话状态和掩码机制

通过这个系统，大语言模型能够动态地调用外部知识库，显著提升推理和问答能力，为构建更智能的AI系统提供了强大的基础架构。
